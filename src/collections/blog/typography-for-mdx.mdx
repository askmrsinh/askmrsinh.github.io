from typing import List, Optional
from functools import lru_cache

class FibonacciCalculator:
    """
    A class for calculating Fibonacci numbers with memoization.
    """

    @staticmethod
    @lru_cache(maxsize=None)
    def calculate(n: int) -> int:
        """
        Calculate the nth Fibonacci number.

        Args:
            n: The position in the Fibonacci sequence

        Returns:
            The Fibonacci number at position n

        Raises:
            ValueError: If n is negative
        """
        if n < 0:
            raise ValueError("Fibonacci sequence is not defined for negative numbers")
        if n <= 1:
            return n
        return FibonacciCalculator.calculate(n-1) + FibonacciCalculator.calculate(n-2)

    @classmethod
    def generate_sequence(cls, length: int) -> List[int]:
        """Generate a Fibonacci sequence of given length."""
        return [cls.calculate(i) for i in range(length)]

# Example usage

```sql title="analytics.sql" showLineNumbers {8-12}
-- User engagement analytics query
WITH user_stats AS (
  SELECT
    u.id,
    u.name,
    u.created_at,
    COUNT(p.id) as post_count,
    COUNT(c.id) as comment_count,
    MAX(p.created_at) as last_post_date
  FROM users u
  LEFT JOIN posts p ON u.id = p.user_id
  LEFT JOIN comments c ON u.id = c.user_id
  WHERE u.created_at >= DATE_SUB(NOW(), INTERVAL 6 MONTH)
  GROUP BY u.id, u.name, u.created_at
),
engagement_scores AS (
  SELECT *,
    (post_count * 3 + comment_count) as engagement_score,
    CASE
      WHEN last_post_date >= DATE_SUB(NOW(), INTERVAL 7 DAY) THEN 'active'
      WHEN last_post_date >= DATE_SUB(NOW(), INTERVAL 30 DAY) THEN 'moderate'
      ELSE 'inactive'
    END as activity_status
  FROM user_stats
)
SELECT
  name,
  engagement_score,
  activity_status,
  RANK() OVER (ORDER BY engagement_score DESC) as engagement_rank
FROM engagement_scores
WHERE engagement_score > 10
ORDER BY engagement_score DESC
LIMIT 20;
```

use std::collections::HashMap;

struct FibonacciCache {
    cache: HashMap<u64, u64>,
}

impl FibonacciCache {
    fn new() -> Self {
        let mut cache = HashMap::new();
        cache.insert(0, 0);
        cache.insert(1, 1);
        Self { cache }
    }

    fn calculate(&mut self, n: u64) -> u64 {
        if let Some(&result) = self.cache.get(&n) {
            return result;
        }

        let result = self.calculate(n - 1) + self.calculate(n - 2);
        self.cache.insert(n, result);
        result
    }
}

fn main() {
    let mut fib = FibonacciCache::new();

    println!("First 15 Fibonacci numbers:");
    for i in 0..15 {
        println!("F({}) = {}", i, fib.calculate(i));
    }
}

import Foundation
import Combine

class NetworkManager: ObservableObject {
    @Published var isLoading = false
    @Published var error: NetworkError?

    private let session = URLSession.shared
    private var cancellables = Set<AnyCancellable>()
    private let baseURL = "https://api.example.com"

    enum NetworkError: Error, LocalizedError {
        case invalidURL
        case noData
        case decodingError

        var errorDescription: String? {
            switch self {
            case .invalidURL: return "Invalid URL"
            case .noData: return "No data received"
            case .decodingError: return "Failed to decode response"
            }
        }
    }

    func fetchData<T: Codable>(
        endpoint: String,
        type: T.Type
    ) -> AnyPublisher<T, NetworkError> {
        guard let url = URL(string: "\(baseURL)/\(endpoint)") else {
            return Fail(error: NetworkError.invalidURL)
                .eraseToAnyPublisher()
        }

        return session.dataTaskPublisher(for: url)
            .map(\.data)
            .decode(type: T.self, decoder: JSONDecoder())
            .mapError { _ in NetworkError.decodingError }
            .eraseToAnyPublisher()
    }
}

    })
    .optional()

// Error handling utilities - collapsible section
class ApiError extends Error {
  constructor(
    message: string,
    public statusCode: number = 500,
    public code?: string
  ) {
    super(message);
    this.name = 'ApiError';
  }
}

const handleApiError = (error: unknown, res: NextApiResponse) => {
  console.error('API Error:', error);

  if (error instanceof ApiError) {
    return res.status(error.statusCode).json({
      error: error.message,
      code: error.code
    });
  }

  if (error instanceof z.ZodError) {
    return res.status(400).json({
      error: 'Validation failed',
      details: error.errors
    });
  }

  return res.status(500).json({
    error: 'Internal server error'
  });
};

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  try {
    // Authentication check
    const session = await authenticate(req);
    if (!session) {
      throw new ApiError('Unauthorized', 401, 'AUTH_REQUIRED');
    }

    // Database operations - collapsible section
    const getUserById = async (id: number) => {
      const user = await prisma.user.findUnique({
        where: { id },
        include: {
          profile: true,
          settings: true,
          posts: {
            take: 5,
            orderBy: { createdAt: 'desc' }
          },
          followers: {
            select: { id: true }
          },
          following: {
            select: { id: true }
          }
        }
      });

      if (!user) {
        throw new ApiError('User not found', 404, 'USER_NOT_FOUND');
      }

      return {
        ...user,
        followersCount: user.followers.length,
        followingCount: user.following.length,
        postsCount: user.posts.length
      };
    };

    const updateUserById = async (id: number, data: any) => {
      const updatedUser = await prisma.user.update({
        where: { id },
        data: {
          ...data,
          updatedAt: new Date()
        },
        include: {
          profile: true,
          settings: true
        }
      });

      return updatedUser;
    };

    // Request handling
    const { id } = querySchema.parse(req.query);

    switch (req.method) {
      case 'GET':
        const user = await getUserById(id);
        return res.status(200).json(user);

      case 'PUT':
        const updateData = userUpdateSchema.parse(req.body);
        const updatedUser = await updateUserById(id, updateData);
        return res.status(200).json(updatedUser);

      case 'DELETE':
        await prisma.user.delete({ where: { id } });
        return res.status(204).end();

      default:
        res.setHeader('Allow', ['GET', 'PUT', 'DELETE']);
        throw new ApiError(`Method ${req.method} not allowed`, 405, 'METHOD_NOT_ALLOWED');
    }
  } catch (error) {
    handleApiError(error, res);
  }
}
